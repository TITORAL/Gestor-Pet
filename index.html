'#f87171' : '#818cf8'; // Vermelho se custo for > 500 }, borderRadius: 8, barThickness: 20 }] }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false } }, scales: { y: { beginAtZero: true, grid: { display: false } }, x: { grid: { display: false } } } } }); } function addDespesa(tipo) { const prefixo = tipo === 'fixa' ? 'fixa' : 'var'; const nome = document.getElementById(`${prefixo}-nome`).value; const valor = parseFloat(document.getElementById(`${prefixo}-valor`).value); if (nome && valor) { const chartId = tipo === 'fixa' ? 'chartFixa' : 'chartVariavel'; charts[chartId].data.labels.push(nome); charts[chartId].data.datasets[0].data.push(valor); charts[chartId].update(); // Limpar inputs document.getElementById(`${prefixo}-nome`).value = ''; document.getElementById(`${prefixo}-valor`).value = ''; } } // Chame no carregamento da página window.onload = () => { updateDates(); charts.chartFixa = initChart('chartFixa', 'Fixas'); charts.chartVariavel = initChart('chartVariavel', 'Variáveis'); if(typeof lucide !== 'undefined') lucide.createIcons(); };
